cmake_minimum_required(VERSION 3.25)
project(NewMapAutoclear VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AUTOCLEAR_BUILD_MOD "Build the Geode mod" ON)
add_library(autoclear_core STATIC src/Solver.cpp)
set_target_properties(autoclear_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(autoclear_core PUBLIC src)
enable_testing()
add_executable(autoclear_tests tests/SolverTests.cpp)
target_link_libraries(autoclear_tests PRIVATE autoclear_core)
add_test(NAME autoclear_tests COMMAND autoclear_tests)
if(AUTOCLEAR_BUILD_MOD)
    if(NOT GEODE_SDK)
        set(GEODE_SDK "$ENV{GEODE_SDK}")
    endif()
    if(NOT EXISTS "${GEODE_SDK}/CMakeLists.txt")
        message(FATAL_ERROR "Set GEODE_SDK or disable AUTOCLEAR_BUILD_MOD")
    endif()
    add_subdirectory("${GEODE_SDK}" geode-sdk)
    add_library(${PROJECT_NAME} SHARED src/main.cpp)
    target_link_libraries(${PROJECT_NAME} PRIVATE autoclear_core)
    setup_geode_mod(${PROJECT_NAME} DONT_INSTALL LINK_TYPE PRIVATE)
endif()
